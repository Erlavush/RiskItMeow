Use this exact setup.

I checked the current Godot docs and Minecraft skin references first. The Godot side you need is: `FileDialog` for filesystem picking, `OS.SYSTEM_DIR_DOWNLOADS` for the default Downloads folder, and runtime image loading turned into a texture with `ImageTexture.create_from_image()`. On the Minecraft side, standard Java-style skins are 64×64 PNG atlases, with two model types: Classic/Wide arms are 4 pixels wide, while Slim arms are 3 pixels wide. Minecraft’s own Java skin import flow also asks you to choose Classic/Wide or Slim when importing, so this demo mirrors that instead of relying on unreliable auto-guessing from the PNG alone. Also, desktop filesystem access through `FileDialog` does not work the same way on web exports. ([Godot Engine documentation][1])

This demo uses a cube-pivot rig instead of smooth skeletal skinning, because that is the cleanest match for Minecraft’s skin atlas. It also uses a 0.6 × 1.8 collision box to match the common player hitbox reference, while keeping the vanilla blocky proportions for the visible model. ([minecraft.fandom.com][2])

Make one empty **Node3D** scene, attach `res://scripts/main.gd` to it, then create these two additional script files.

static func create_cuboid(size_px: Vector3, uv_map: Dictionary, inflate_px: float = 0.0) -> ArrayMesh:
	var hx := size_px.x * 0.5 + inflate_px
	var hy := size_px.y * 0.5 + inflate_px
	var hz := size_px.z * 0.5 + inflate_px

	var st := SurfaceTool.new()
	st.begin(Mesh.PRIMITIVE_TRIANGLES)

	# Front (+Z)
	_add_quad(
		st,
		Vector3(-hx, hy, hz),
		Vector3(-hx, -hy, hz),
		Vector3(hx, -hy, hz),
		Vector3(hx, hy, hz),
		uv_map["front"],
		Vector3.FORWARD
	)

	# Back (-Z)
	_add_quad(
		st,
		Vector3(hx, hy, -hz),
		Vector3(hx, -hy, -hz),
		Vector3(-hx, -hy, -hz),
		Vector3(-hx, hy, -hz),
		uv_map["back"],
		Vector3.BACK
	)

	# Right (+X)
	_add_quad(
		st,
		Vector3(hx, hy, hz),
		Vector3(hx, -hy, hz),
		Vector3(hx, -hy, -hz),
		Vector3(hx, hy, -hz),
		uv_map["right"],
		Vector3.RIGHT
	)

	# Left (-X)
	_add_quad(
		st,
		Vector3(-hx, hy, -hz),
		Vector3(-hx, -hy, -hz),
		Vector3(-hx, -hy, hz),
		Vector3(-hx, hy, hz),
		uv_map["left"],
		Vector3.LEFT
	)

	# Top (+Y)
	_add_quad(
		st,
		Vector3(-hx, hy, -hz),
		Vector3(-hx, hy, hz),
		Vector3(hx, hy, hz),
		Vector3(hx, hy, -hz),
		uv_map["top"],
		Vector3.UP
	)

	# Bottom (-Y)
	_add_quad(
		st,
		Vector3(-hx, -hy, hz),
		Vector3(-hx, -hy, -hz),
		Vector3(hx, -hy, -hz),
		Vector3(hx, -hy, hz),
		uv_map["bottom"],
		Vector3.DOWN
	)

	return st.commit()

static func _add_quad(
	st: SurfaceTool,
	a: Vector3,
	b: Vector3,
	c: Vector3,
	d: Vector3,
	uv_rect: Rect2,
	normal: Vector3
) -> void:
	var uv := _rect_to_uv_points(uv_rect)

	_add_vertex(st, a, uv[0], normal)
	_add_vertex(st, b, uv[1], normal)
	_add_vertex(st, c, uv[2], normal)

	_add_vertex(st, a, uv[0], normal)
	_add_vertex(st, c, uv[2], normal)
	_add_vertex(st, d, uv[3], normal)

static func _add_vertex(st: SurfaceTool, vertex: Vector3, uv: Vector2, normal: Vector3) -> void:
	st.set_normal(normal)
	st.set_uv(uv)
	st.add_vertex(vertex)

static func _rect_to_uv_points(rect: Rect2) -> Array[Vector2]:
	var tex_size := 64.0
	var u0 := rect.position.x / tex_size
	var v0 := rect.position.y / tex_size
	var u1 := (rect.position.x + rect.size.x) / tex_size
	var v1 := (rect.position.y + rect.size.y) / tex_size

	return [
		Vector2(u0, v0), # top-left
		Vector2(u0, v1), # bottom-left
		Vector2(u1, v1), # bottom-right
		Vector2(u1, v0), # top-right
	]

```gdscript
# res://scripts/minecraft_player.gd
class_name MinecraftPlayer
extends CharacterBody3D

const PIXEL_SIZE := 1.8 / 32.0
const MOVE_SPEED := 5.0
const JUMP_VELOCITY := 5.25

const BODY_OVERLAY_INFLATE := 0.25
const HEAD_OVERLAY_INFLATE := 0.5

enum SkinModel {
	CLASSIC,
	SLIM,
}

var gravity: float = 9.8
var skin_model: int = SkinModel.CLASSIC

var skin_texture: Texture2D
var base_material: StandardMaterial3D
var overlay_material: StandardMaterial3D
var overlay_visible := false

var visual_root: Node3D
var body_pivot: Node3D
var head_pivot: Node3D
var right_arm_pivot: Node3D
var left_arm_pivot: Node3D
var right_leg_pivot: Node3D
var left_leg_pivot: Node3D

var camera_rig: Node3D
var camera_3d: Camera3D

var overlay_parts: Array[MeshInstance3D] = []
var walk_time := 0.0

func _ready() -> void:
	gravity = float(ProjectSettings.get_setting("physics/3d/default_gravity"))
	_ensure_input_map()
	_setup_collision()
	_setup_camera()

	skin_texture = _make_default_texture()
	_rebuild_materials()
	_rebuild_visuals()

func _process(delta: float) -> void:
	if camera_3d != null:
		camera_3d.look_at(global_position + Vector3(0, 1.2, 0), Vector3.UP)
	_animate_model(delta)

func _physics_process(delta: float) -> void:
	if not is_on_floor():
		velocity.y -= gravity * delta

	var input_x := Input.get_action_strength("move_right") - Input.get_action_strength("move_left")
	var input_z := Input.get_action_strength("move_back") - Input.get_action_strength("move_forward")
	var input_dir := Vector2(input_x, input_z)
	if input_dir.length() > 1.0:
		input_dir = input_dir.normalized()

	var desired := Vector3(input_dir.x, 0.0, input_dir.y) * MOVE_SPEED
	var accel := 20.0 if is_on_floor() else 8.0

	velocity.x = move_toward(velocity.x, desired.x, accel * delta)
	velocity.z = move_toward(velocity.z, desired.z, accel * delta)

	if is_on_floor() and Input.is_action_just_pressed("jump"):
		velocity.y = JUMP_VELOCITY

	move_and_slide()

	if visual_root != null and input_dir.length() > 0.01:
		visual_root.rotation.y = lerp_angle(
			visual_root.rotation.y,
			atan2(input_dir.x, input_dir.y),
			12.0 * delta
		)

func set_skin_model(new_model: int) -> void:
	if new_model != SkinModel.CLASSIC and new_model != SkinModel.SLIM:
		return

	if skin_model == new_model:
		return

	skin_model = new_model
	_rebuild_visuals()

func load_skin_from_path(path: String, requested_model: int) -> bool:
	var image := Image.load_from_file(path)
	if image.get_width() != 64 or image.get_height() != 64:
		return false

	skin_model = requested_model
	skin_texture = ImageTexture.create_from_image(image)
	overlay_visible = true

	_rebuild_materials()
	_rebuild_visuals()
	return true

func _setup_collision() -> void:
	var collider := CollisionShape3D.new()
	var shape := BoxShape3D.new()
	shape.size = Vector3(0.6, 1.8, 0.6)
	collider.shape = shape
	collider.position = Vector3(0, 0.9, 0)
	add_child(collider)

func _setup_camera() -> void:
	camera_rig = Node3D.new()
	camera_rig.name = "CameraRig"
	camera_rig.position = Vector3(0, 2.35, 4.25)
	add_child(camera_rig)

	camera_3d = Camera3D.new()
	camera_3d.name = "Camera3D"
	camera_3d.current = true
	camera_3d.fov = 60.0
	camera_rig.add_child(camera_3d)

func _rebuild_materials() -> void:
	base_material = _make_skin_material(skin_texture)
	overlay_material = _make_skin_material(skin_texture)

func _rebuild_visuals() -> void:
	if visual_root != null:
		visual_root.free()

	overlay_parts.clear()

	visual_root = Node3D.new()
	visual_root.name = "VisualRoot"
	visual_root.scale = Vector3.ONE * PIXEL_SIZE
	add_child(visual_root)

	body_pivot = Node3D.new()
	body_pivot.name = "BodyPivot"
	body_pivot.position = Vector3(0, 24, 0)
	visual_root.add_child(body_pivot)

	head_pivot = Node3D.new()
	head_pivot.name = "HeadPivot"
	head_pivot.position = Vector3(0, 24, 0)
	visual_root.add_child(head_pivot)

	right_arm_pivot = Node3D.new()
	right_arm_pivot.name = "RightArmPivot"
	visual_root.add_child(right_arm_pivot)

	left_arm_pivot = Node3D.new()
	left_arm_pivot.name = "LeftArmPivot"
	visual_root.add_child(left_arm_pivot)

	right_leg_pivot = Node3D.new()
	right_leg_pivot.name = "RightLegPivot"
	right_leg_pivot.position = Vector3(-2, 12, 0)
	visual_root.add_child(right_leg_pivot)

	left_leg_pivot = Node3D.new()
	left_leg_pivot.name = "LeftLegPivot"
	left_leg_pivot.position = Vector3(2, 12, 0)
	visual_root.add_child(left_leg_pivot)

	var arm_width := 3.0 if skin_model == SkinModel.SLIM else 4.0
	right_arm_pivot.position = Vector3(-4.0 - arm_width * 0.5, 24, 0)
	left_arm_pivot.position = Vector3(4.0 + arm_width * 0.5, 24, 0)

	_build_body()
	_build_head()
	_build_arms(arm_width)
	_build_legs()

	_set_overlay_visible(overlay_visible)

func _build_body() -> void:
	var body_mesh := McSkinMesh.create_cuboid(Vector3(8, 12, 4), McSkinMesh.body_uv(true), 0.0)
	var body_overlay_mesh := McSkinMesh.create_cuboid(Vector3(8, 12, 4), McSkinMesh.body_uv(false), BODY_OVERLAY_INFLATE)

	_add_mesh(body_pivot, body_mesh, Vector3(0, -6, 0), base_material, false)
	_add_mesh(body_pivot, body_overlay_mesh, Vector3(0, -6, 0), overlay_material, true)

func _build_head() -> void:
	var head_mesh := McSkinMesh.create_cuboid(Vector3(8, 8, 8), McSkinMesh.head_uv(true), 0.0)
	var head_overlay_mesh := McSkinMesh.create_cuboid(Vector3(8, 8, 8), McSkinMesh.head_uv(false), HEAD_OVERLAY_INFLATE)

	_add_mesh(head_pivot, head_mesh, Vector3(0, 4, 0), base_material, false)
	_add_mesh(head_pivot, head_overlay_mesh, Vector3(0, 4, 0), overlay_material, true)

func _build_arms(arm_width: float) -> void:
	var slim := skin_model == SkinModel.SLIM

	var right_arm_mesh := McSkinMesh.create_cuboid(
		Vector3(arm_width, 12, 4),
		McSkinMesh.right_arm_uv(true, slim),
		0.0
	)
	var right_arm_overlay_mesh := McSkinMesh.create_cuboid(
		Vector3(arm_width, 12, 4),
		McSkinMesh.right_arm_uv(false, slim),
		BODY_OVERLAY_INFLATE
	)

	var left_arm_mesh := McSkinMesh.create_cuboid(
		Vector3(arm_width, 12, 4),
		McSkinMesh.left_arm_uv(true, slim),
		0.0
	)
	var left_arm_overlay_mesh := McSkinMesh.create_cuboid(
		Vector3(arm_width, 12, 4),
		McSkinMesh.left_arm_uv(false, slim),
		BODY_OVERLAY_INFLATE
	)

	_add_mesh(right_arm_pivot, right_arm_mesh, Vector3(0, -6, 0), base_material, false)
	_add_mesh(right_arm_pivot, right_arm_overlay_mesh, Vector3(0, -6, 0), overlay_material, true)

	_add_mesh(left_arm_pivot, left_arm_mesh, Vector3(0, -6, 0), base_material, false)
	_add_mesh(left_arm_pivot, left_arm_overlay_mesh, Vector3(0, -6, 0), overlay_material, true)

func _build_legs() -> void:
	var right_leg_mesh := McSkinMesh.create_cuboid(Vector3(4, 12, 4), McSkinMesh.right_leg_uv(true), 0.0)
	var right_leg_overlay_mesh := McSkinMesh.create_cuboid(Vector3(4, 12, 4), McSkinMesh.right_leg_uv(false), BODY_OVERLAY_INFLATE)

	var left_leg_mesh := McSkinMesh.create_cuboid(Vector3(4, 12, 4), McSkinMesh.left_leg_uv(true), 0.0)
	var left_leg_overlay_mesh := McSkinMesh.create_cuboid(Vector3(4, 12, 4), McSkinMesh.left_leg_uv(false), BODY_OVERLAY_INFLATE)

	_add_mesh(right_leg_pivot, right_leg_mesh, Vector3(0, -6, 0), base_material, false)
	_add_mesh(right_leg_pivot, right_leg_overlay_mesh, Vector3(0, -6, 0), overlay_material, true)

	_add_mesh(left_leg_pivot, left_leg_mesh, Vector3(0, -6, 0), base_material, false)
	_add_mesh(left_leg_pivot, left_leg_overlay_mesh, Vector3(0, -6, 0), overlay_material, true)

func _add_mesh(
	parent: Node3D,
	mesh: ArrayMesh,
	local_position_px: Vector3,
	material: Material,
	is_overlay: bool
) -> MeshInstance3D:
	var mesh_instance := MeshInstance3D.new()
	mesh_instance.mesh = mesh
	mesh_instance.position = local_position_px
	mesh_instance.material_override = material
	parent.add_child(mesh_instance)

	if is_overlay:
		overlay_parts.append(mesh_instance)

	return mesh_instance

func _set_overlay_visible(value: bool) -> void:
	for part in overlay_parts:
		part.visible = value

func _animate_model(delta: float) -> void:
	if visual_root == null:
		return

	var horizontal_speed := Vector2(velocity.x, velocity.z).length()
	var moving := horizontal_speed > 0.1

	if moving:
		walk_time += delta * 10.0 * clamp(horizontal_speed / MOVE_SPEED, 0.5, 1.5)
		var arm_swing := sin(walk_time) * 0.8
		var leg_swing := sin(walk_time) * 0.8

		right_arm_pivot.rotation.x = arm_swing
		left_arm_pivot.rotation.x = -arm_swing
		right_leg_pivot.rotation.x = -leg_swing
		left_leg_pivot.rotation.x = leg_swing
	else:
		right_arm_pivot.rotation.x = lerp(right_arm_pivot.rotation.x, 0.0, 8.0 * delta)
		left_arm_pivot.rotation.x = lerp(left_arm_pivot.rotation.x, 0.0, 8.0 * delta)
		right_leg_pivot.rotation.x = lerp(right_leg_pivot.rotation.x, 0.0, 8.0 * delta)
		left_leg_pivot.rotation.x = lerp(left_leg_pivot.rotation.x, 0.0, 8.0 * delta)

	head_pivot.rotation.x = lerp(
		head_pivot.rotation.x,
		deg_to_rad(2.0) * sin(Time.get_ticks_msec() * 0.002),
		4.0 * delta
	)

func _make_skin_material(texture: Texture2D) -> StandardMaterial3D:
	var material := StandardMaterial3D.new()
	material.albedo_texture = texture
	material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	material.texture_filter = BaseMaterial3D.TEXTURE_FILTER_NEAREST
	material.cull_mode = BaseMaterial3D.CULL_DISABLED
	return material

func _make_default_texture() -> Texture2D:
	var image := Image.create(64, 64, false, Image.FORMAT_RGBA8)
	image.fill(Color(1, 1, 1, 1))
	return ImageTexture.create_from_image(image)

func _ensure_input_map() -> void:
	_add_key_if_missing("move_forward", Key.W)
	_add_key_if_missing("move_back", Key.S)
	_add_key_if_missing("move_left", Key.A)
	_add_key_if_missing("move_right", Key.D)
	_add_key_if_missing("jump", Key.SPACE)

func _add_key_if_missing(action_name: String, key: Key) -> void:
	if not InputMap.has_action(action_name):
		InputMap.add_action(action_name)

	for existing in InputMap.action_get_events(action_name):
		if existing is InputEventKey and existing.physical_keycode == key:
			return

	var event := InputEventKey.new()
	event.physical_keycode = key
	InputMap.action_add_event(action_name, event)

	choose_button.pressed.connect(_on_choose_skin_pressed)
	vbox.add_child(choose_button)

	info_label = Label.new()
	info_label.text = "Pick the correct arm type, then choose a 64x64 PNG skin."
	info_label.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	vbox.add_child(info_label)

	file_dialog = FileDialog.new()
	file_dialog.access = FileDialog.ACCESS_FILESYSTEM
	file_dialog.file_mode = FileDialog.FILE_MODE_OPEN_FILE
	file_dialog.use_native_dialog = true
	file_dialog.filters = PackedStringArray(["*.png ; PNG Image"])
	file_dialog.file_selected.connect(_on_file_selected)
	canvas.add_child(file_dialog)

func _on_choose_skin_pressed() -> void:
	var downloads_path := OS.get_system_dir(OS.SYSTEM_DIR_DOWNLOADS)
	if downloads_path != "":
		file_dialog.current_dir = downloads_path

	file_dialog.popup_centered_ratio(0.75)

func _on_model_selected(index: int) -> void:
	var selected_id := model_picker.get_item_id(index)
	player.set_skin_model(selected_id)

func _on_file_selected(path: String) -> void:
	var selected_model := model_picker.get_selected_id()
	var ok := player.load_skin_from_path(path, selected_model)

	if ok:
		info_label.text = "Loaded skin: %s" % path.get_file()
	else:
		info_label.text = "Invalid skin. Use a 64x64 PNG made for Minecraft skins."
```

How to use it:

1. Make an empty `Node3D` scene.
2. Attach `res://scripts/main.gd`.
3. Save and run.
4. Pick **Classic / Wide** or **Slim** first.
5. Click **Choose 64x64 Skin PNG**.
6. It will open the OS file picker and point at Downloads by default on desktop platforms. Godot documents `SYSTEM_DIR_DOWNLOADS` for this, and `FileDialog.current_dir` is treated as a hint for native dialogs, so some operating systems may not honor it exactly. ([Godot Engine documentation][3])

This supports standard 64×64 skins with both base and outer layers. It does not cover Bedrock-style 128×128 skins or capes. Minecraft’s own skin references distinguish the standard 64×64 format and the classic vs slim arm models, so that is the exact scope this setup targets. ([minecraft.fandom.com][4])

The next sensible step is adding first-person arms or importing the same rig into a Blockbench-authored character scene.
